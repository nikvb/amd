# app_amd_ws test scenarios - one call per line, '|' separated, '#' comments.
# run.sh generates a [farside-<name>] and an [amdside-<name>] context per row
# (extension pattern _XXXX = call counter) and runs the rows whose tags match the mode.
#
# columns:
#  name     scenario name = context suffix and VID suffix; [a-z0-9_] only, unique
#  tags     self  = plumbing scenario, runs in --selftest and in full mode
#           amd   = needs the built app_amd_ws.so (full mode only)
#           db    = proves the MySQL code path; SKIP when the module was built without MySQL
#           probe = not run on its own; used by the unload/reload checks
#  count    number of simultaneous originates (all within 1 s); 1 for a single call
#  farside  behaviour subroutine [fs-<farside>] in asterisk/extensions.conf.in:
#           speech8 speech8-mix speech12 speech1200-hold speech1200-hangup beeps speech1500-hangup noaudio silence8 record3 dial-noanswer
#  mock     mock server path written to the control file before the call
#           ('-' = leave as is; the mock is not involved)
#  amdside  what the amdside runs between the T0/T1 timestamps:
#           ws:<timeout_ms>,<playfile>,<options>   -> AMD_WS(127.0.0.1,${MOCK_PORT},${VID},<...>)
#           wsdead:<timeout_ms>,<playfile>,<options> -> same but ${DEAD_PORT} (nothing listens)
#           wstls:<timeout_ms>,<playfile>,<options>  -> same but ${TLS_PORT} (wss mock; put 's' in the options; SKIP without openssl)
#           wshole:<timeout_ms>,<playfile>,<options> -> same but ${BLACKHOLE_PORT} (accepts TCP, never answers the upgrade)
#           wsraw:<full argument string>            -> AMD_WS(<full argument string>)
#           app:App1(args) && App2(args) ...        -> those dialplan applications verbatim
#  post     dialplan apps to run after amdside returned, before Hangup ('-' = none)
#  status   expected AMDSTATUS   ('*' = do not check)
#  cause    expected AMDCAUSE    ('*' = do not check; '~REGEX' = grep -E match, e.g. ~^NOAUDIODATA-[0-9]+$;
#           for MACHINE the cause is the server's raw reply text, exactly as amd.py sets it)
#           Every AMD_WS row is also checked for AMDSTATS =~ ^[0-9]+-[0-9]+-[0-9]+-[0-9]+$ with
#           run_time (field 1) within +/-150 ms of AMDELAPSED and audio_ms*16 == bytes.
#  min/max  bounds for the wall-clock ms the amdside step took (T1 - T0);
#           NOTE: the farside Answer() returns up to 500 ms after the amdside
#           started when the amdside sends no media, so audio-driven results
#           land ~500 ms later than the AMD clock (AMDELAPSED) suggests.
#  asserts  comma list, evaluated after the status/cause/timing checks:
#           proto                      protocol_test.py config,schedule,bytes,eof,close (+ no phone / caller_id keys)
#           proto:c1;c2[;phone=N][;country=C][;callerid=X][;schedule=L]   explicit protocol_test checks (';' separated);
#                                      without phone=/country= the config must have no phone keys, without callerid= no caller_id;
#                                      checks: config schedule bytes eof eofmarks close chunks result noresult
#           chunks=N chunks>=N chunks<=N   number of binary chunks the mock received
#           noconn                     the mock has NO connection record for this VID
#           resp~TEXT resp=TEXT        decoded AMDRESPONSE contains / equals TEXT
#           elapsed>=N elapsed<=N      AMDELAPSED (module's own clock) bounds
#           heard>THR heard<THR        sox RMS of <VID>-heard.wav (what the callee heard)
#           heard[start:len]>THR / <THR   RMS of a segment (seconds), file time 0 = farside Answer
#           heard_dur<S heard_dur>S    duration of <VID>-heard.wav in seconds
#           mix>THR                    RMS of <VID>-mix.wav (both directions)
#           log~REGEX nolog~REGEX      full log (grep -E) contains / lacks REGEX; %VID% %NUM% %NAME% are replaced
#           cli~REGEX                  'amd_ws show settings' (grep -E) after the call ended
#           mockvid=TEXT               the mock parsed exactly TEXT as the VID (%VID% replaced; JSON escaping round trip)
#           (assert tokens are comma separated: no commas inside a token; use "." in a regex)
#           alive                      asterisk still answers the CLI afterwards (always checked too)
#           Bursts (count > 1) get log~/nolog~/cli~/alive with %VID% = the first call, plus a check
#           that the mock saw exactly one connection per VID.
#  Upper bounds (max_ms, elapsed<=) are multiplied by TEST_SLOW_FACTOR (default 1); lower bounds are not.
#  The AMD_WS channel in log~ patterns is  Local/%NUM%@farside-%NAME%-[0-9a-f]+;1  (anchors the line to THIS call).
#
